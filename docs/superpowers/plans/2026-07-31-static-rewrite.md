# Static Rewrite Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the Blazor Server web app with a fully static, client-side Persian↔Gregorian converter (horological-minimal design, system dark mode, analog Spring-Drive clock, 366 bilingual stoic quotes) served by nginx, deployed directly to the Hetzner host over SSH (Coolify has been removed).

**Architecture:** New `Site/` directory: Vite + TypeScript, no runtime framework. The Persian calendar algorithm is ported to TS and proven bit-exact against .NET's `System.Globalization.PersianCalendar` via test vectors generated by the `Console` project. `Web/` (Blazor) is deleted after the static site is verified live.

**Tech Stack:** TypeScript 5.x (strict), Vite 7, Vitest 3, Playwright, nginx:alpine, .NET 10 (Lib/Console/Tests unchanged in role).

**Spec:** `docs/superpowers/specs/2026-07-31-static-rewrite-design.md`

## Global Constraints

- JS < 20 KB gzipped; CSS < 15 KB gzipped (fonts excluded; they are separate woff2 files).
- No runtime JS framework; no external origins — CSP `default-src 'self'`; all fonts self-hosted.
- **Supported conversion range = parity-verified range:** Persian years **1200–1500** inclusive (≈ Gregorian 1821–2122). Vectors cover every day of that range. (Amends spec §5/§6.1 — see Task 2 Step 6.)
- Bilingual EN/FA. FA mode: `<html lang="fa" dir="rtl">`, Persian-Indic digits everywhere, Vazirmatn font.
- Dark/light purely via `prefers-color-scheme`. No toggle.
- Quote of the day keyed by **Persian day-of-year** (1 Farvardin = 1, 30 Esfand = 366).
- nginx listens on **8080** (Caddy on the host proxies `time.automated.ink` → `time-app:8080`).
- **Host facts (verified 2026-07-31):** Hetzner `root@159.69.19.71`; Caddy compose at `/opt/caddy` (external docker network `edge`); app compose at `/opt/time/docker-compose.yml` running host-local image `time-app:current`, `restart: unless-stopped`. Deploys = build image in CI → `docker save | ssh docker load` → `docker compose up -d`. The old Coolify webhook deploy is dead (run 30623349923, curl exit 35).
- Number inputs keep Western digits even in FA mode (native `type="number"` constraint); all rendered text, dates, and clock numerals use Persian-Indic digits in FA.
- .NET builds use `-warnaserror`. TS uses `"strict": true`.
- Commit at the end of every task (checkpoint commits approved by Ali for this plan). Do **not** push until Task 14 says so — the first push after Task 12 switches the production Dockerfile.
- Playwright E2E: behavioral assertions only (no pixel-baseline assertions — they are OS-dependent); screenshots are saved as artifacts for manual review.

```ts
export interface PersianDateParts { py: number; pm: number; pd: number }
export interface GregorianDateParts { gy: number; gm: number; gd: number }

export const PERSIAN_YEAR_MIN = 1200;
export const PERSIAN_YEAR_MAX = 1500;

// Break years of the Birashk arithmetic Jalaali calendar (jalaali-js, MIT).
const BREAKS = [
  -61, 9, 38, 199, 426, 686, 756, 818, 1111, 1181, 1210,
  1635, 2060, 2097, 2192, 2262, 2324, 2394, 2456, 3178,
];

const div = (a: number, b: number): number => Math.trunc(a / b);
const mod = (a: number, b: number): number => a - Math.trunc(a / b) * b;

interface JalCal { leap: number; gy: number; march: number }

function jalCal(py: number): JalCal {
  const last = BREAKS[BREAKS.length - 1] as number;
  if (py < (BREAKS[0] as number) || py >= last) {
    throw new RangeError(`Persian year out of algorithm range: ${py}`);
  }
  const gy = py + 621;
  let leapJ = -14;
  let jp = BREAKS[0] as number;
  let jump = 0;
  for (let i = 1; i < BREAKS.length; i += 1) {
    const jm = BREAKS[i] as number;
    jump = jm - jp;
    if (py < jm) break;
    leapJ += div(jump, 33) * 8 + div(mod(jump, 33), 4);
    jp = jm;
  }
  let n = py - jp;
  leapJ += div(n, 33) * 8 + div(mod(n, 33) + 3, 4);
  if (mod(jump, 33) === 4 && jump - n === 4) leapJ += 1;
  const leapG = div(gy, 4) - div((div(gy, 100) + 1) * 3, 4) - 150;
  const march = 20 + leapJ - leapG;
  if (jump - n < 6) n = n - jump + div(jump + 4, 33) * 33;
  let leap = mod(mod(n + 1, 33) - 1, 4);
  if (leap === -1) leap = 4;
  return { leap, gy, march };
}

function g2d(gy: number, gm: number, gd: number): number {
  let d = div((gy + div(gm - 8, 6) + 100100) * 1461, 4)
    + div(153 * mod(gm + 9, 12) + 2, 5)
    + gd - 34840408;
  d = d - div(div(gy + 100100 + div(gm - 8, 6), 100) * 3, 4) + 752;
  return d;
}

function d2g(jdn: number): GregorianDateParts {
  let j = 4 * jdn + 139361631;
  j = j + div(div(4 * jdn + 183187720, 146097) * 3, 4) * 4 - 3908;
  const i = div(mod(j, 1461), 4) * 5 + 308;
  const gd = div(mod(i, 153), 5) + 1;
  const gm = mod(div(i, 153), 12) + 1;
  const gy = div(j, 1461) - 100100 + div(8 - gm, 6);
  return { gy, gm, gd };
}

function j2d(py: number, pm: number, pd: number): number {
  const r = jalCal(py);
  return g2d(r.gy, 3, r.march) + (pm - 1) * 31 - div(pm, 7) * (pm - 7) + pd - 1;
}

function d2j(jdn: number): PersianDateParts {
  const gy = d2g(jdn).gy;
  let py = gy - 621;
  const r = jalCal(py);
  const jdn1f = g2d(gy, 3, r.march);
  let k = jdn - jdn1f;
  if (k >= 0) {
    if (k <= 185) {
      return { py, pm: 1 + div(k, 31), pd: mod(k, 31) + 1 };
    }
    k -= 186;
  } else {
    py -= 1;
    k += 179;
    if (r.leap === 1) k += 1;
  }
  return { py, pm: 7 + div(k, 30), pd: mod(k, 30) + 1 };
}

export function isLeapPersianYear(py: number): boolean {
  return jalCal(py).leap === 0;
}

export function daysInPersianMonth(py: number, pm: number): number {
  if (pm <= 6) return 31;
  if (pm <= 11) return 30;
  return isLeapPersianYear(py) ? 30 : 29;
}

export function daysInGregorianMonth(gy: number, gm: number): number {
  const lengths = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31];
  const isLeapG = (gy % 4 === 0 && gy % 100 !== 0) || gy % 400 === 0;
  if (gm === 2 && isLeapG) return 29;
  return lengths[gm - 1] as number;
}

function assertSupportedPersianYear(py: number): void {
  if (py < PERSIAN_YEAR_MIN || py > PERSIAN_YEAR_MAX) {
    throw new RangeError(`Persian year out of supported range: ${py}`);
  }
}

export function toGregorian(p: PersianDateParts): GregorianDateParts {
  assertSupportedPersianYear(p.py);
  if (p.pm < 1 || p.pm > 12) throw new RangeError(`Invalid Persian month: ${p.pm}`);
  if (p.pd < 1 || p.pd > daysInPersianMonth(p.py, p.pm)) {
    throw new RangeError(`Invalid Persian day: ${p.pd}`);
  }
  return d2g(j2d(p.py, p.pm, p.pd));
}

export function toPersian(g: GregorianDateParts): PersianDateParts {
  if (g.gm < 1 || g.gm > 12) throw new RangeError(`Invalid Gregorian month: ${g.gm}`);
  if (g.gd < 1 || g.gd > daysInGregorianMonth(g.gy, g.gm)) {
    throw new RangeError(`Invalid Gregorian day: ${g.gd}`);
  }
  const p = d2j(g2d(g.gy, g.gm, g.gd));
  assertSupportedPersianYear(p.py);
  return p;
}

export function persianDayOfYear(p: PersianDateParts): number {
  return p.pm <= 6 ? (p.pm - 1) * 31 + p.pd : 186 + (p.pm - 7) * 30 + p.pd;
}
```

```csharp
using System.Globalization;
using System.Text.Json;

namespace Vectors;

public sealed record Vector(int Py, int Pm, int Pd, int Gy, int Gm, int Gd);

public static class VectorGenerator
{
    private static readonly JsonSerializerOptions JsonOptions =
        new() { PropertyNamingPolicy = JsonNamingPolicy.CamelCase };

    public static IReadOnlyList<Vector> Generate(int fromPersianYear, int toPersianYear)
    {
        var calendar = new PersianCalendar();
        var vectors = new List<Vector>();
        for (var py = fromPersianYear; py <= toPersianYear; py++)
        {
            for (var pm = 1; pm <= 12; pm++)
            {
                var days = calendar.GetDaysInMonth(py, pm);
                for (var pd = 1; pd <= days; pd++)
                {
                    var g = calendar.ToDateTime(py, pm, pd, 0, 0, 0, 0);
                    vectors.Add(new Vector(py, pm, pd, g.Year, g.Month, g.Day));
                }
            }
        }
        return vectors;
    }

    public static void WriteJson(string path, int fromPersianYear, int toPersianYear)
    {
        var directory = Path.GetDirectoryName(Path.GetFullPath(path));
        if (!string.IsNullOrEmpty(directory)) Directory.CreateDirectory(directory);
        File.WriteAllText(path, JsonSerializer.Serialize(Generate(fromPersianYear, toPersianYear), JsonOptions));
    }
}

import {
  daysInGregorianMonth, daysInPersianMonth,
  PERSIAN_YEAR_MAX, PERSIAN_YEAR_MIN,
  toGregorian, toPersian,
  type GregorianDateParts, type PersianDateParts,
} from '../calendar/persian';

export type Side = 'persian' | 'gregorian';
export interface RawFields { y: string; m: string; d: string }

export const GREGORIAN_YEAR_MIN = 1822;
export const GREGORIAN_YEAR_MAX = 2121;

export type ConversionOutcome =
  | { status: 'ok'; persian: PersianDateParts; gregorian: GregorianDateParts }
  | { status: 'incomplete' }
  | { status: 'error'; errorKey: 'errYearRange' | 'errMonth' | 'errDay';
      params: Record<string, number>; side: Side };

function parseField(raw: string): number | null {
  if (raw.trim() === '') return null;
  const n = Number(raw);
  return Number.isInteger(n) ? n : null;
}

export function convertFrom(side: Side, fields: RawFields): ConversionOutcome {
  const y = parseField(fields.y);
  const m = parseField(fields.m);
  const d = parseField(fields.d);
  if (y === null || m === null || d === null) return { status: 'incomplete' };

  const [yearMin, yearMax] = side === 'persian'
    ? [PERSIAN_YEAR_MIN, PERSIAN_YEAR_MAX]
    : [GREGORIAN_YEAR_MIN, GREGORIAN_YEAR_MAX];
  if (y < yearMin || y > yearMax) {
    return { status: 'error', errorKey: 'errYearRange', params: { min: yearMin, max: yearMax }, side };
  }
  if (m < 1 || m > 12) {
    return { status: 'error', errorKey: 'errMonth', params: {}, side };
  }
  const daysInMonth = side === 'persian' ? daysInPersianMonth(y, m) : daysInGregorianMonth(y, m);
  if (d < 1 || d > daysInMonth) {
    return { status: 'error', errorKey: 'errDay', params: { month: m, days: daysInMonth, year: y }, side };
  }

  if (side === 'persian') {
    const persian = { py: y, pm: m, pd: d };

import { convertFrom, type Side } from './converter-core';
import {
  formatTemplate, GREGORIAN_MONTHS, PERSIAN_MONTHS, STRINGS, type Lang,
} from '../i18n/strings';
import type { GregorianDateParts, PersianDateParts } from '../calendar/persian';

interface Refs {
  persian: { y: HTMLInputElement; m: HTMLInputElement; d: HTMLInputElement };
  gregorian: { y: HTMLInputElement; m: HTMLInputElement; d: HTMLInputElement };
  error: HTMLElement;
}

function byId<T extends HTMLElement>(id: string): T {
  const el = document.getElementById(id);
  if (!el) throw new Error(`Missing element #${id}`);
  return el as T;
}

export function setupConverter(getLang: () => Lang): {
  prefill: (p: PersianDateParts, g: GregorianDateParts) => void;
  rerender: () => void;
} {
  const refs: Refs = {
    persian: { y: byId('p-y'), m: byId('p-m'), d: byId('p-d') },
    gregorian: { y: byId('g-y'), m: byId('g-m'), d: byId('g-d') },
    error: byId('converter-error'),
  };
  let lastEdited: Side = 'persian';

  const setSide = (side: Side, p: PersianDateParts, g: GregorianDateParts): void => {
    if (side === 'persian') {
      refs.persian.y.value = String(p.py);
      refs.persian.m.value = String(p.pm);
      refs.persian.d.value = String(p.pd);
    } else {
      refs.gregorian.y.value = String(g.gy);
      refs.gregorian.m.value = String(g.gm);
      refs.gregorian.d.value = String(g.gd);
    }
  };

  const showError = (message: string | null): void => {
    refs.error.hidden = message === null;
    refs.error.textContent = message ?? '';
  };

  const run = (source: Side): void => {
    lastEdited = source;
    const lang = getLang();
    const f = refs[source];
    const outcome = convertFrom(source, { y: f.y.value, m: f.m.value, d: f.d.value });
    if (outcome.status === 'ok') {
      setSide(source === 'persian' ? 'gregorian' : 'persian', outcome.persian, outcome.gregorian);
      showError(null);
      return;
    }
    if (outcome.status === 'incomplete') {
      showError(null);
      return;
    }
    const params = { ...outcome.params };
    if (outcome.errorKey === 'errDay') {
      const months = outcome.side === 'persian' ? PERSIAN_MONTHS[lang] : GREGORIAN_MONTHS[lang];
      // month name substitutes the numeric index for the message
      (params as Record<string, string | number>)['month'] = months[(outcome.params['month'] as number) - 1] as string;
    }
    showError(formatTemplate(STRINGS[lang][outcome.errorKey], params, lang));
  };

  for (const side of ['persian', 'gregorian'] as const) {
    for (const input of Object.values(refs[side])) {
      input.addEventListener('input', () => run(side));
    }
  }

  return {
    prefill: (p, g) => {
      setSide('persian', p, g);
      setSide('gregorian', p, g);
      showError(null);
    },
    rerender: () => run(lastEdited),
  };
}
```

import { handAngles, msSinceLocalMidnight } from './clock-math';

const TICKS = Array.from({ length: 60 }, (_, i) => {
  const angle = (i / 60) * 360;
  const isHour = i % 5 === 0;
  const r1 = isHour ? 88 : 92;
  return `<line x1="100" y1="${100 - r1}" x2="100" y2="${100 - 96}"
    transform="rotate(${angle} 100 100)"
    stroke="currentColor" stroke-width="${isHour ? 1.6 : 0.6}"
    opacity="${isHour ? 0.9 : 0.35}" />`;
}).join('');

const CLOCK_SVG = `
<svg viewBox="0 0 200 200" width="260" height="260" role="img" aria-label="Analog clock" class="clock">
  <circle cx="100" cy="100" r="98" fill="none" stroke="currentColor" stroke-width="1" opacity="0.25" />
  ${TICKS}
  <g id="hand-hour"><line x1="100" y1="100" x2="100" y2="52"
    stroke="currentColor" stroke-width="4" stroke-linecap="round" /></g>
  <g id="hand-minute"><line x1="100" y1="100" x2="100" y2="30"
    stroke="currentColor" stroke-width="2.4" stroke-linecap="round" /></g>
  <g id="hand-second"><line x1="100" y1="112" x2="100" y2="22"
    class="second-hand" stroke-width="1" stroke-linecap="round" /></g>
  <circle cx="100" cy="100" r="3" fill="currentColor" />
</svg>`;

export function createClock(mount: HTMLElement): () => void {
  mount.innerHTML = CLOCK_SVG;
  const hour = mount.querySelector<SVGGElement>('#hand-hour');
  const minute = mount.querySelector<SVGGElement>('#hand-minute');
  const second = mount.querySelector<SVGGElement>('#hand-second');
  if (!hour || !minute || !second) throw new Error('Clock SVG hands missing');

  const reduced = window.matchMedia('(prefers-reduced-motion: reduce)');
  let rafId = 0;
  let intervalId = 0;

  const render = (): void => {
    const a = handAngles(msSinceLocalMidnight(new Date()));
    hour.setAttribute('transform', `rotate(${a.hour} 100 100)`);
    minute.setAttribute('transform', `rotate(${a.minute} 100 100)`);
    second.setAttribute('transform', `rotate(${a.second} 100 100)`);
  };

  const loop = (): void => {
    render();
    rafId = requestAnimationFrame(loop);
  };

  const stop = (): void => {
    cancelAnimationFrame(rafId);
    window.clearInterval(intervalId);
    rafId = 0;
    intervalId = 0;
  };

  const start = (): void => {
    stop();
    if (reduced.matches) {
      render();
      intervalId = window.setInterval(render, 1000);
    } else {
      loop();
    }
  };

  const onVisibility = (): void => {
    if (document.hidden) stop();
    else start();
  };

  document.addEventListener('visibilitychange', onVisibility);
  reduced.addEventListener('change', start);
  start();

  return () => {
    stop();
    document.removeEventListener('visibilitychange', onVisibility);
    reduced.removeEventListener('change', start);
  };
}
